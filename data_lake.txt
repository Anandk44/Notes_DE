WITH lag_cte AS
(
    SELECT *,
           LAG(login_date) OVER
           (
               PARTITION BY user
               ORDER BY login_date
           ) AS prev_date
    FROM logins
),
flag_cte AS
(
    SELECT *,
           CASE
               WHEN prev_date IS NULL THEN 1
               WHEN DATEDIFF(day, prev_date, login_date) > 1 THEN 1
               ELSE 0
           END AS new_group
    FROM lag_cte
),
group_cte AS
(
    SELECT *,
           SUM(new_group) OVER
           (
               PARTITION BY user
               ORDER BY login_date
               ROWS UNBOUNDED PRECEDING
           ) AS grp
    FROM flag_cte
)
SELECT user,
       MIN(login_date) AS start_date,
       MAX(login_date) AS end_date,
       COUNT(*) AS days
FROM group_cte
GROUP BY user, grp
ORDER BY start_date;
